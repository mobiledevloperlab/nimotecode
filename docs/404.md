<script setup>
import { onMounted } from 'vue'

// Corresponding-source archives are served from Cloudflare R2, not from this
// site, and hosts that do not implement public/_redirects (GitHub Pages among
// them) answer a request for one with this page. Forward such a request to the
// object that actually holds it, so links written against the old in-repo
// location keep resolving. Client-only: the archive path is only knowable in
// the browser.
onMounted(() => {
  const archive = window.location.pathname.match(
    /^\/opensource\/local-linux\/(.+\.(?:tar\.zst|tar\.gz|tgz|tar\.xz))$/
  )
  if (archive) {
    window.location.replace(`https://opensource-downloads.nimotecode.com/local-linux/${archive[1]}`)
  }
})
</script>
